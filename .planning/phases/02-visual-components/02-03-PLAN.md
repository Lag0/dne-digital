    ```typescript
    import Image from 'next/image';

    interface PhotoCardProps {
      src: string;
      alt: string;
    }

    /**
     * Card da foto do estudante com proporcao correta via next/image fill mode.
     * Parent externo tem w-1/2 e o container pai tem h-[280px] — fill herda essas dimensoes.
     */
    export const PhotoCard = ({ src, alt }: PhotoCardProps) => (
      <div className="bg-white rounded-xl p-1.5 shadow-sm w-1/2 relative">
        {/* Inner div precisa de position:relative e dimensoes para o fill funcionar */}
        <div className="relative w-full h-full rounded-lg overflow-hidden">
          <Image
            src={src}
            alt={alt}
            fill
            sizes="195px"
            className="object-cover"
          />
        </div>
      </div>
    );
    ```

    REGRAS CRITICAS:
    - Parent interno DEVE ter `position: relative` (classe `relative`) — sem isso, Image fill colapsa para 0px
    - NAO usar `width` e `height` fixos no Image — usar apenas `fill`
    - `sizes="195px"` — metade de 390px (largura do card)
    - `overflow-hidden` no inner div para o `object-cover` ser respeitado visualmente
    - `h-full` no inner div herda a altura do parent externo que herda do container h-[280px]
    - Named export (nao default) — padrao do projeto
